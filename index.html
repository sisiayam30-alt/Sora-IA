import numpy as np import pandas as pd import ta from sklearn.ensemble import RandomForestClassifier class AITradingBot: def __init__(self, initial_stake=1.0, martingale_multiplier=2.0): self.initial_stake = initial_stake self.stake = initial_stake self.multiplier = martingale_multiplier self.current_step = 0 # Saina IA (Random Forest Classifier) self.model = RandomForestClassifier(n_estimators=100, random_state=42) self.is_trained = False def generate_sample_data(self): """ Mamorona data vonjimaika hanaovana test (mety soloina data tena izy avy eo) """ np.random.seed(42) data = { 'close': np.cumsum(np.random.randn(100) * 0.5) + 100, 'high': np.cumsum(np.random.randn(100) * 0.5) + 101, 'low': np.cumsum(np.random.randn(100) * 0.5) + 99, } return pd.DataFrame(data) def calculate_indicators(self, df): """ Manao kajy ireo indicators rehetra nangatahinao """ df['MA50'] = ta.trend.sma_indicator(df['close'], window=10) # Natao fohy kely ho an'ny test df['RSI'] = ta.momentum.rsi(df['close'], window=14) macd = ta.trend.MACD(df['close']) df['MACD'] = macd.macd() df['MACD_signal'] = macd.macd_signal() bollinger = ta.volatility.BollingerBands(df['close'], window=20, window_dev=2.0) df['BB_high'] = bollinger.bollinger_hband() df['BB_low'] = bollinger.bollinger_lband() return df def train_ai(self, df): """ Mampianatra ny Saina IA miorina amin'ny angona """ df = self.calculate_indicators(df) df.dropna(inplace=True) X = df[['RSI', 'MACD', 'MA50', 'BB_high', 'BB_low']] y = np.where(df['close'].shift(-1) > df['close'], 1, 0) if len(X) > 1: self.model.fit(X[:-1], y[:-1]) self.is_trained = True print(">>> [IA] Voatofana tsara ny Saina IA!") def analyze_market(self, df, duration): """ Manao analyse ary mamoaka sipe miaraka amin'ny Ora voafidy """ df = self.calculate_indicators(df) last_row = df.iloc[-1:] if not self.is_trained: # Fepetra fototra (Rule-based) raha mbola tsy voatofana ny IA rsi = last_row['RSI'].values[0] if rsi < 35: return "CALL (MIVIDY)", 0.75 elif rsi > 65: return "PUT (MIVAROTRA)", 0.75 return "WAIT (MIANDRY)", 0.50 X_current = last_row[['RSI', 'MACD', 'MA50', 'BB_high', 'BB_low']] probs = self.model.predict_proba(X_current)[0] prob_put, prob_call = probs[0], probs[1] if prob_call > 0.65: return "CALL (MIVIDY)", prob_call elif prob_put > 0.65: return "PUT (MIVAROTRA)", prob_put else: return "WAIT (MIANDRY)", max(prob_call, prob_put) def update_martingale(self, won: bool): """ Mitantana ny vola (Martingale) """ if won: print(">>> Vokatra: NANDRESY 🎉 -> Miverina amin'ny voalohany.") self.stake = self.initial_stake self.current_step = 0 else: self.current_step += 1 self.stake *= self.multiplier print(f">>>> Vokatra: RESY ❌ -> Martingale dingana {self.current_step}: Vola vaovao = ${self.stake}") # --- RUNNER PROGRAM --- if __name__ == "__main__": bot = AITradingBot(initial_stake=1.0, martingale_multiplier=2.0) # 1. Mamorona data sy manofana ny IA sample_df = bot.generate_sample_data() bot.train_ai(sample_df) print("\n==============================================") print("🤖 BOT ANALYSE & AI SIGNAL GENERATOR VONONA 🤖") print("==============================================") while True: print("\n--- SAFIDY ORA (TIMEFRAME) ---") duration = input("Safidio ny fotoana (5s, 10s, 15s, 30s, 1M, 2M, 3M) na 'q' hivoahana: ") if duration.lower() == 'q': print("Misaotra tamin'ny fampiasana ny bot!") break if duration in ["5s", "10s", "15s", "30s", "1M", "2M", "3M"]: print(f"Manao analyse amin'ny ora [{duration}]...") # Maka data vaovao (na simulated) df = bot.generate_sample_data() signal, confidence = bot.analyze_market(df, duration) print(f"----------------------------------------------") print(f"📊 VOKATRA ANALYSE: {signal}") print(f"📈 Fahatokisana (Confidence): {confidence * 100:.2f}%") print(f"💰 Vola ampiasaina (Stake): ${bot.stake}") print(f"----------------------------------------------") # Fanandramana valiny (Simulation win/loss ho an'ny Martingale) res = input("Inona ny valin'ity trade ity? (n = Nandresy, r = Resy): ").lower() if res == 'n': bot.update_martingale(won=True) elif res == 'r': bot.update_martingale(won=False) else: print("⚠️ Tsy mety io safidy io. Safidio tsara ao anaty lisitra.")
